createHearts();

window.addEventListener('click', (e) => {
if(e.target.closest('.popup-card')) return;
triggerFingerBurst(e.clientX, e.clientY);
});

window.addEventListener('mousemove', (e) => {
const container = document.getElementById("mainContainer");
const xAxis = (window.innerWidth / 2 - e.clientX) / 25;
const yAxis = (window.innerHeight / 2 - e.clientY) / 25;
container.style.transform = rotateY(${xAxis}deg) rotateX(${yAxis}deg);
});
});

// 2. Background Ambient Drift Generator
function createHearts() {
const container = document.getElementById("particles-container");
const heartsArr = ["❤️", "💖", "💕", "🌸"];

setInterval(() => {
const particle = document.createElement("div");
particle.className = "particle";
particle.innerText = heartsArr[Math.floor(Math.random() * heartsArr.length)];

particle.style.left = Math.random() * 100 + "vw";
particle.style.setProperty('--travel-x', (Math.random() * 100 - 50) + "px");
particle.style.fontSize = (Math.random() * 12 + 12) + "px";
particle.style.animationDuration = (Math.random() * 3 + 5) + "s";

container.appendChild(particle);
setTimeout(() => { particle.remove(); }, 8000);
}, 600);
}

// 3. Fingertip Click Heart Burst Engine
function triggerFingerBurst(x, y) {
if(!x && !y) return;
const burstPool = ["❤️", "💖", "💕", "✨"];

for (let i = 0; i < 5; i++) {
const heart = document.createElement('div');
heart.className = 'burst-heart';
heart.innerText = burstPool[Math.floor(Math.random() * burstPool.length)];

heart.style.left = x + 'px';
heart.style.top = y + 'px';

const angle = Math.random() * Math.PI * 2;
const distance = Math.random() * 45 + 25;
const mx = Math.cos(angle) * distance + 'px';
const my = Math.sin(angle) * distance + 'px';

heart.style.setProperty('--mx', mx);
heart.style.setProperty('--my', my);

document.body.appendChild(heart);
setTimeout(() => { heart.remove(); }, 600);
}
}

// Handles precise multi-screen logic and custom screen 4 timeline
function changeScreen(screenNum) {
document.querySelectorAll('.screen').forEach(screen => {
screen.classList.remove('active');
});

const targetScreen = document.getElementById('screen' + screenNum);
targetScreen.classList.add('active');

// entering screen 4: Optimized precisely to a 2.8s window setup
if (screenNum === 4) {
const dateEl = document.getElementById("targetDate");
const originalGroupEl = document.getElementById("originalGroup");
const hiddenGroupEl = document.getElementById("hiddenGroup");
const nextBtn = document.getElementById("ticketNextBtn");

dateEl.classList.remove("struck");
originalGroupEl.classList.remove("dimmed");
hiddenGroupEl.classList.remove("reveal");
nextBtn.style.visibility = "hidden";

// Step 1: Wait exactly 2.8 seconds displaying only the original date layout
setTimeout(() => {
// Step 2: Animate line cross strike out
dateEl.classList.add("struck");

// Step 3: Wait for strike transition to complete, then dim original text and drop below elements
setTimeout(() => {
originalGroupEl.classList.add("dimmed");
hiddenGroupEl.classList.add("reveal");
nextBtn.style.visibility = "visible";
}, 800);

}, 2800);
}
}

// 4. Custom Overlay Modal Controller Action
function openPopup() {
document.getElementById("popupAlert").classList.add("active");
}
